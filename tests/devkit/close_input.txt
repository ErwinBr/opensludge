# Close the tab of each kind of document as its X does, in the middle of a
# frame in which the tab has drawn, and see that the Dev Kit goes on. The
# editors with pictures once freed their textures while the frame still
# had to draw them.
open ${FIXTURES}/floor-edited.flo
wait 3
documents 1
close
wait 2
documents 0

open ${FIXTURES}/zbuffer-edited.zbu
wait 3
close
wait 2
documents 0

open ${EXAMPLES}/VerbCoin/ego/alien.duc
wait 3
close
wait 2
documents 0

open ${FIXTURES}/german-edited.tra
wait 3
close
wait 2
documents 0

# Two at once, closing the one behind first with Ctrl+W after the front.

wait 2

wait 3
documents 2

wait 2
close
wait 2
documents 0
